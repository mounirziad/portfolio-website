Project Overview

Arctic Invasion is a modular, scalable mobile tower defense game developed with Unity, featuring strategic tower placement, dynamic enemy waves, and persistent progression systems. As Lead Programmer, I architected core gameplay systems that balance strategic depth with mobile-friendly accessibility.

This project demonstrates production-ready mobile game development practices, including performance-optimized pathfinding, responsive touch controls, and scalable architecture supporting extensive content updates and monetization systems.

Core Technical Systems

🏗️ Modular Tower System

Extensible tower architecture supporting multiple types, upgrades, and placement mechanics.

C#
public class TowerSystem : MonoBehaviour { [SerializeField] private TowerData[] towerTypes; [SerializeField] private LayerMask placementLayer; [SerializeField] private CurrencySystem currency; private Tower selectedTower; private bool isPlacingTower; public void StartTowerPlacement(TowerData towerData) { if (currency.CanAfford(towerData.cost)) { selectedTower = Instantiate(towerData.prefab).GetComponent(); selectedTower.Initialize(towerData); isPlacingTower = true; ShowValidPlacementAreas(); } } private void Update() { if (isPlacingTower) { HandleTowerPlacement(); } } private void HandleTowerPlacement() { Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition); if (Physics.Raycast(ray, out RaycastHit hit, 100f, placementLayer)) { selectedTower.transform.position = hit.point; if (Input.GetMouseButtonDown(0) && IsValidPlacement(hit.point)) { PlaceTower(hit.point); } } } private void PlaceTower(Vector3 position) { currency.Spend(selectedTower.TowerData.cost); selectedTower.OnPlaced(); isPlacingTower = false; HidePlacementAreas(); } }

🎯 Enemy Wave Management

Dynamic wave system with spawn patterns, pathfinding, and behavior trees.

C#
public class WaveManager : MonoBehaviour { [SerializeField] private WaveData[] waves; [SerializeField] private Transform[] spawnPoints; [SerializeField] private PathfindingGrid pathfindingGrid; private int currentWave = 0; private int enemiesRemaining = 0; private bool waveInProgress = false; public void StartNextWave() { if (!waveInProgress && currentWave < waves.Length) { StartCoroutine(SpawnWave(waves[currentWave])); currentWave++; } } private IEnumerator SpawnWave(WaveData waveData) { waveInProgress = true; enemiesRemaining = waveData.totalEnemies; foreach (var spawnGroup in waveData.spawnGroups) { for (int i = 0; i < spawnGroup.count; i++) { SpawnEnemy(spawnGroup.enemyType, GetSpawnPoint()); yield return new WaitForSeconds(spawnGroup.spawnInterval); } } yield return new WaitUntil(() => enemiesRemaining <= 0); waveInProgress = false; OnWaveCompleted?.Invoke(currentWave); } private void SpawnEnemy(EnemyType enemyType, Transform spawnPoint) { var enemy = Instantiate(enemyType.prefab, spawnPoint.position, Quaternion.identity); var enemyAI = enemy.GetComponent(); enemyAI.Initialize(pathfindingGrid, OnEnemyDefeated); } private void OnEnemyDefeated(EnemyAI enemy) { enemiesRemaining--; currency.Add(enemy.Reward); } } [System.Serializable] public class WaveData { public string waveName; public SpawnGroup[] spawnGroups; public int totalEnemies; public float waveReward; }

⚔️ Barracks & Unit System

Dynamic unit spawning from player-placed barracks with strategic AI behaviors.

C#
public class Barracks : Tower { [SerializeField] private UnitData unitData; [SerializeField] private float spawnInterval = 3f; [SerializeField] private int maxUnits = 5; private List spawnedUnits = new List(); private float spawnTimer; protected override void Update() { base.Update(); if (spawnedUnits.Count < maxUnits) { spawnTimer += Time.deltaTime; if (spawnTimer >= spawnInterval) { SpawnUnit(); spawnTimer = 0f; } } } private void SpawnUnit() { var unit = Instantiate(unitData.prefab, transform.position, Quaternion.identity); var unitComponent = unit.GetComponent(); unitComponent.Initialize(unitData, this); spawnedUnits.Add(unitComponent); unitComponent.OnUnitDestroyed += OnUnitDestroyed; } private void OnUnitDestroyed(Unit unit) { spawnedUnits.Remove(unit); unit.OnUnitDestroyed -= OnUnitDestroyed; } public Vector3 GetRallyPoint() { // Calculate strategic position based on enemy proximity return transform.position + transform.forward * 2f; } } public class Unit : MonoBehaviour { private UnitData data; private Barracks homeBarracks; private EnemyAI target; public void Initialize(UnitData unitData, Barracks barracks) { data = unitData; homeBarracks = barracks; FindTarget(); } private void FindTarget() { // Find closest enemy within attack range var enemies = FindObjectsOfType(); float closestDistance = Mathf.Infinity; foreach (var enemy in enemies) { float distance = Vector3.Distance(transform.position, enemy.transform.position); if (distance < data.attackRange && distance < closestDistance) { target = enemy; closestDistance = distance; } } if (target != null) { StartCoroutine(AttackRoutine()); } } }

💰 Economy & Progression

Persistent currency system with in-game purchases and reward logic.

C#
public class CurrencySystem : MonoBehaviour { [SerializeField] private int startingCurrency = 100; private int currentCurrency; private const string CURRENCY_KEY = "PlayerCurrency"; public int CurrentCurrency => currentCurrency; public event Action OnCurrencyChanged; private void Start() { LoadCurrency(); } public void Add(int amount) { currentCurrency += amount; SaveCurrency(); OnCurrencyChanged?.Invoke(currentCurrency); } public bool Spend(int amount) { if (CanAfford(amount)) { currentCurrency -= amount; SaveCurrency(); OnCurrencyChanged?.Invoke(currentCurrency); return true; } return false; } public bool CanAfford(int amount) { return currentCurrency >= amount; } private void LoadCurrency() { currentCurrency = PlayerPrefs.GetInt(CURRENCY_KEY, startingCurrency); } private void SaveCurrency() { PlayerPrefs.SetInt(CURRENCY_KEY, currentCurrency); PlayerPrefs.Save(); } } public class ShopSystem : MonoBehaviour { [SerializeField] private ShopItem[] availableItems; [SerializeField] private CurrencySystem currency; [SerializeField] private UIManager uiManager; public void PurchaseItem(ShopItem item) { if (currency.Spend(item.cost)) { ApplyItemEffect(item); uiManager.ShowPurchaseSuccess(item); } else { uiManager.ShowInsufficientFunds(); } } private void ApplyItemEffect(ShopItem item) { switch (item.itemType) { case ItemType.TowerUnlock: TowerManager.Instance.UnlockTower(item.towerData); break; case ItemType.CurrencyBoost: currency.Add(item.boostAmount); break; case ItemType.Upgrade: ApplyUpgrade(item.upgradeData); break; } } } [System.Serializable] public class ShopItem { public string itemName; public int cost; public Sprite icon; public ItemType itemType; public TowerData towerData; public int boostAmount; public UpgradeData upgradeData; }
