Project Overview

Mechanisoul is an advanced Unity project showcasing sophisticated game architecture patterns and modular systems design. As Lead Programmer at DreamStatic Studios, I engineered extensible character controllers, combat frameworks, and AI systems that demonstrate production-ready code quality and scalable architecture.

This project highlights modern Unity development practices, including ScriptableObject-driven data design, event-driven architecture, and animation-integrated gameplay systems that maintain clean separation of concerns while delivering responsive, polished gameplay.

Core Technical Systems

🎮 Modular Character Controller

Extensible player controller supporting multiple mechanics with clean state management.

C#
public class ModularCharacterController : MonoBehaviour { [SerializeField] private CharacterConfig config; [SerializeField] private PlayerInput input; [SerializeField] private Camera playerCamera; private CharacterState currentState; private Vector3 velocity; private bool isGrounded; public void TransitionToState(CharacterState newState) { currentState?.Exit(); currentState = newState; currentState.Enter(); } private void Update() { currentState?.Update(); ApplyMovement(); HandleAnimation(); } private void ApplyMovement() { Vector2 moveInput = input.Move; Vector3 movement = new Vector3(moveInput.x, 0, moveInput.y); movement = playerCamera.transform.TransformDirection(movement); movement.y = 0; if (movement.magnitude > 1) movement.Normalize(); velocity.x = movement.x * config.MoveSpeed; velocity.z = movement.z * config.MoveSpeed; ApplyGravity(); controller.Move(velocity * Time.deltaTime); } }

⚔️ ScriptableObject Combat Framework

Data-driven combat system supporting multiple weapon types and attack patterns.

C#
[CreateAssetMenu(menuName = "Combat/Weapon")] public class WeaponData : ScriptableObject { [Header("Weapon Properties")] public string weaponName; public WeaponType weaponType; public float baseDamage; public float attackSpeed; public float range; [Header("Visual & Audio")] public GameObject weaponPrefab; public AudioClip attackSound; public AnimationClip attackAnimation; [Header("Combo System")] public AttackCombo[] attackCombos; public void ExecuteAttack(Transform attacker, int comboIndex = 0) { var attack = attackCombos[comboIndex]; StartCoroutine(PerformAttackSequence(attacker, attack)); } private IEnumerator PerformAttackSequence(Transform attacker, AttackCombo attack) { foreach (var action in attack.attackActions) { yield return new WaitForSeconds(action.delay); action.Execute(attacker, this); } } } [System.Serializable] public struct AttackCombo { public string comboName; public AttackAction[] attackActions; public float comboResetTime; }

🎯 Advanced Targeting System

Zelda-inspired lock-on system with smooth camera transitions and input management.

C#
public class LockOnSystem : MonoBehaviour { [SerializeField] private float lockOnRange = 10f; [SerializeField] private float lockOnRadius = 3f; [SerializeField] private LayerMask targetableLayers; private Transform currentTarget; private CameraController cameraController; private List availableTargets = new List(); public void ToggleLockOn() { if (currentTarget != null) ReleaseLockOn(); else AcquireLockOn(); } private void AcquireLockOn() { availableTargets.Clear(); FindPotentialTargets(); if (availableTargets.Count > 0) { currentTarget = GetClosestTarget(); cameraController.SetLockOnTarget(currentTarget); OnLockOnAcquired?.Invoke(currentTarget); } } private void FindPotentialTargets() { Collider[] hits = Physics.OverlapSphere(transform.position, lockOnRange, targetableLayers); foreach (var hit in hits) { Vector3 directionToTarget = (hit.transform.position - transform.position).normalized; if (Vector3.Dot(transform.forward, directionToTarget) > 0.5f) { availableTargets.Add(hit.transform); } } } private Transform GetClosestTarget() { Transform closest = null; float closestDistance = Mathf.Infinity; foreach (var target in availableTargets) { float distance = Vector3.Distance(transform.position, target.position); if (distance < closestDistance) { closest = target; closestDistance = distance; } } return closest; } }

🤖 AI Behavior Tree System

Modular AI framework with state-driven behaviors and configurable decision making.

C#
public class AIBehaviorTree : MonoBehaviour { [SerializeField] private AIConfig config; [SerializeField] private AIBlackboard blackboard; private BTNode rootNode; private AIState currentState; private Transform player; private void Start() { player = GameObject.FindGameObjectWithTag("Player").transform; InitializeBehaviorTree(); } private void InitializeBehaviorTree() { // Build behavior tree structure rootNode = new BTSelector( new BTSequence( new BTCheckPlayerInRange(blackboard, config.attackRange), new BTSetState(blackboard, AIState.Attacking), new BTAttackPlayer(config) ), new BTSequence( new BTCheckPlayerVisible(blackboard), new BTSetState(blackboard, AIState.Chasing), new BTChasePlayer(config, blackboard) ), new BTPatrol(config, blackboard) ); } private void Update() { UpdateBlackboard(); rootNode?.Evaluate(); UpdateState(); } private void UpdateBlackboard() { blackboard.PlayerPosition = player.position; blackboard.DistanceToPlayer = Vector3.Distance(transform.position, player.position); blackboard.CanSeePlayer = CheckLineOfSight(); } } [System.Serializable] public class AIConfig { public float patrolSpeed = 2f; public float chaseSpeed = 4f; public float attackRange = 3f; public float sightRange = 10f; public float[] patrolWaitTimes = { 2f, 4f }; public Transform[] patrolPoints; }

Technical Architecture & Design Patterns

  • Modular System Design: Separated character, combat, and AI into independent, reusable systems
  • ScriptableObject Architecture: Data-driven design for weapons, abilities, and AI configurations
  • Event-Driven Communication: Decoupled systems using UnityEvents and custom delegates
  • State Pattern Implementation: Clean state management for character behaviors and AI
  • Animation Integration: Event-driven animation system with hitbox synchronization
  • Extensible Combat Framework: Support for multiple weapon types and combo systems
  • Behavior Tree AI: Modular AI system with configurable decision making
